3 · Google Apps Script fetcher — automatic daily + matchday

Runs on Google's infrastructure (no Cloudflare, no Vercel egress problems). Two time-triggers:

  • Daily 05:00 UTC (+ manual menu item): pulls your entry + picks → squad-push
  • Sat & Sun hourly 14:00–18:00 UTC: pulls live points → live-push
  1. Go to script.google.com → New project.
  2. Paste the script below. Set ENGINE_BASE and TOKEN at the top.
  3. Add triggers: clock icon → syncSquadDaily, Time-driven, Day timer 5–6am. Add a second trigger for syncLiveMatchday, Week timer Sat+Sun 2–6pm.
// ==== CONFIG ====
var ENGINE_BASE = "PASTE_ENGINE_URL"; // e.g. https://your-app.vercel.app
var TOKEN = "PASTE_SYNC_PUSH_TOKEN";
var ENTRY_ID = 1234567;               // your FPL entry id

// ==== Daily squad sync (time-trigger ~05:00) ====
function syncSquadDaily() { syncSquad(); }

// ==== Matchday live points (Sat/Sun hourly 14:00-18:00) ====
function syncLiveMatchday() {
  var now = new Date();
  var day = now.getUTCDay();          // 6=Sat 0=Sun
  if (day !== 6 && day !== 0) return;
  var gw = currentGw();
  if (!gw) return;
  var live = jget("/api/event/" + gw + "/live/");
  var els = (live.elements || []).map(function (e) {
    return {
      element_id: e.id,
      points: e.stats && e.stats.total_points || 0,
      minutes: e.stats && e.stats.minutes || 0
    };
  });
  post("/api/v1/sync/live-push", { gameweek: gw, elements: els });
}

// ==== Shared helpers — v2.5.3 truth: fetch current + next GW, prefer latest transfer ====
function syncSquad() {
  var entry = jget("/api/entry/" + ENTRY_ID + "/");
  var currentGw = (entry.entry && entry.entry.current_event) || entry.current_event || 1;
  var nextGw = currentGw + 1;
  // Try to refine nextGw via bootstrap when reachable (same logic as bookmarklet).
  try {
    var bs = jget("/api/bootstrap-static/");
    var evs = (bs && bs.events) || [];
    var best = null, bestTime = null, now = Date.now();
    for (var i = 0; i < evs.length; i++) {
      var ev = evs[i];
      if (!ev || !ev.deadline_time) continue;
      var dl = Date.parse(ev.deadline_time);
      if (isNaN(dl) || dl <= now) continue;
      if (bestTime === null || dl < bestTime) { bestTime = dl; best = ev.id; }
    }
    if (best !== null && best !== currentGw) nextGw = best;
    else if (best === currentGw) nextGw = currentGw + 1;
  } catch (e) { /* bootstrap optional */ }
  var picksCur = null, picksNext = null;
  try { picksCur = jget("/api/entry/" + ENTRY_ID + "/event/" + currentGw + "/picks/"); } catch (e) { picksCur = null; }
  try { if (nextGw && nextGw !== currentGw) picksNext = jget("/api/entry/" + ENTRY_ID + "/event/" + nextGw + "/picks/"); } catch (e) { picksNext = null; }
  var chosen = picksCur, chosenGw = currentGw;
  if (picksCur && picksNext) {
    var idsCur = (picksCur.picks || []).map(function (p) { return p.element; }).sort().join(",");
    var idsNext = (picksNext.picks || []).map(function (p) { return p.element; }).sort().join(",");
    if (idsCur !== idsNext) { chosen = picksNext; chosenGw = nextGw; }
  } else if (!picksCur && picksNext) { chosen = picksNext; chosenGw = nextGw; }
  if (!chosen) throw new Error("No picks for GW" + currentGw + " nor GW" + nextGw);
  post("/api/v1/sync/squad-push", {
    entry_id: ENTRY_ID,
    entry_name: (entry.entry && entry.entry.name) || entry.name || null,
    gameweek: chosenGw,
    picks_gw: chosenGw,
    bank: (chosen.entry_history && chosen.entry_history.bank) || 0,
    transfers: {
      made: (chosen.entry_history && chosen.entry_history.event_transfers) || 0,
      cost: (chosen.entry_history && chosen.entry_history.event_transfers_cost) || 0
    },
    picks: (chosen.picks || []).map(function (p) {
      return {
        element_id: p.element, position: p.position,
        element_type: p.element_type,
        is_captain: !!p.is_captain, is_vice: !!p.is_vice
      };
    })
  });
}

function bootstrap() {
  var r = UrlFetchApp.fetch("https://fantasy.premierleague.com/api/bootstrap-static/", {
    headers: { "User-Agent": "Mozilla/5.0" }, muteHttpExceptions: true });
  return JSON.parse(r.getContentText());
}
function currentGw() {
  var events = bootstrap().events || [];
  var live = events.filter(function (e) { return e.is_current; })[0];
  if (!live) live = events.filter(function (e) { return e.is_next; })[0];
  return live ? live.id : null;
}
function jget(path) {
  var r = UrlFetchApp.fetch("https://fantasy.premierleague.com" + path, {
    headers: { "User-Agent": "Mozilla/5.0", "Accept": "application/json" },
    muteHttpExceptions: true });
  if (r.getResponseCode() !== 200) throw new Error(path + " -> " + r.getResponseCode());
  return JSON.parse(r.getContentText());
}
function post(path, body) {
  var r = UrlFetchApp.fetch(ENGINE_BASE + path, {
    method: "post", contentType: "application/json",
    headers: { "Authorization": "Bearer " + TOKEN },
    payload: JSON.stringify(body), muteHttpExceptions: true });
  Logger.log(path + " -> " + r.getResponseCode() + " " + r.getContentText());
}

// ==== Manual menu (Extensions > FPL Sync > Run all) ====
function onOpen() {
  try {
    SpreadsheetApp.getUi().createMenu("FPL Sync")
      .addItem("Sync squad now", "syncSquad")
      .addItem("Push live points now", "syncLiveMatchday")
      .addToUi();
  } catch (e) { /* standalone script has no UI */ }
}
